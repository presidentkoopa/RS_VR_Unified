// ============================================================
// from RS_Hands  (MODELDEF.txt)

// Rigged VR hands, as a fist weapon.
//
// Frame layout mirrors RS_Main's RS_GH_Fist, which drives its swing off sprite
// frames: A -> model frame 0, B -> 1, C -> 4, D -> 6. The animation baked into
// these models is authored to those exact indices -- open at 0, closing through
// 1, fist driving forward at 4, follow through at 6.
//
// NOAUTOREVERSE matters here. The engine mirrors whichever model lands on the
// left physical controller so one weapon mesh can serve both hands. These are
// two purpose-built left and right meshes, so that mirror would flip an
// already-correct hand. RS_GH_Fist needs the mirror (Scale -1 mainhand, +1
// offhand) precisely because it is one mesh; we do not.



// ---------------------------------------------------------------------
// ALWAYS-ON HANDS
//
// Separate from the fist weapon above, and they have to be. A weapon owns
// its hand slot, so a fist model is only ever drawn when nothing else is
// equipped -- it cannot show a hand and a gun on the same controller.
// These ride their own psprite layers instead, so they are visible no
// matter what is held.
//
// BaseFrame, not FrameIndex, and it must come last. BaseFrame is the only
// thing that registers into BaseSpriteModelFrames and sets hasmodel, which
// is what a +DECOUPLEDANIMATIONS actor's model lookup reads. Without it the
// psprite falls through to the sprite path and nothing is drawn.
// ---------------------------------------------------------------------

// There is no hand_right.iqm any more, and its absence is deliberate. The file
// that used to ship here was never referenced by either block below, and it was
// STALE: built from an older pose list carrying 11 baked poses where hand_left
// carries 20. Anything that switched to it would have silently lost nine poses
// and shifted every index above 10. A 600KB file that looks current and is not
// is worse than no file at all. If a genuinely separate right-hand mesh is ever
// wanted, rebuild it from hand_final.fbx with tools/tools_export_poses.py
// rather than resurrecting that one.

// Both hands use hand_left.iqm, and the main hand mirrors it with a negative
// X scale. That is not a workaround, it is the convention every other weapon
// model here follows (RS_GH_Fist, VanillaVRPlus: Scale -1 one hand, +1 the
// other).
//
// The reason: measured signed volume of the two meshes is -1288.6 and +1288.6
// -- identical magnitude, opposite sign, so they are true mirrors and the pair
// is geometrically correct. But this HUD path flips normals unconditionally, so
// the negative-winding mesh is the one that shades correctly and the positive
// one renders inside out. Driving the mirror from scale instead keeps shading
// right, because the renderer watches for a negative scale product and flips
// face culling to match.
Model RS_HandIdleMain
{
	Path "models/hands"
	Model 0 "hand_left.iqm"
	Skin 0 "hand_basecolor.png"
	Scale -1.0 1.0 1.0
	Offset 0.0 -18.0 -12.0
	UseHandOffsets
	NOAUTOREVERSE
	PitchOffset 90.0
	RollOffset 0.0
	AngleOffset 0.0
	BaseFrame
}

Model RS_HandIdleOff

// ===========================================================================
// WORLD-ACTOR HANDS. Same wire as the world M9: MDL_FOLLOWMAINHAND/OFFHAND make
// the model take its transform from GetWeaponTransform at DRAW time, instead of
// from wherever script last put the actor at 35Hz.
//
// Scale 0.01. MEASURED, not derived from a target size, and it is one clean
// number because the two render paths differ by exactly one factor of 100.
//
// BOTH PATHS START FROM THE SAME MATRIX. GetWeaponTransform (hw_vrmodes.cpp:1483)
// hands back the controller pose with mat->scale(vr_vunits_per_meter) already
// applied (vk_openxrdevice.cpp:5926), so downstream of it mesh space is METRES on
// both paths. They then diverge in one place and one place only:
//
//   HUD path   RenderHUDModel applies a further scale(0.01f) (models.cpp:934-937),
//              documented there as the model-unit conversion every hud model needs.
//   WORLD path ObjectToWorldMatrix applies no such factor -- MODELDEF Scale goes
//              straight into step 3 (models.cpp:849).
//
// So a world Scale of 0.01 draws a followed model at EXACTLY the size the same
// mesh draws at on a psprite with Scale 1.0. That is the whole derivation, and it
// needs no anatomy: the psprite hands are the size that was signed off in a
// headset, and this matches them by construction.
//
// Cross-check against the mesh. hand_left.iqm bbox extent measured from its
// POSITION vertex array is 14.9882 x 9.0510 x 32.9533, longest axis 32.9533.
//   psprite: 32.9533 * 1.0 (MODELDEF) * 1.0 (rs_hand_scale -> psp.scale,
//            rs_hands.zs:392-396 -> pspScale, models.cpp:985) * 0.01 = 0.3295 m
//   world:   32.9533 * 0.01 * 1.0 (rs_hw_*_scale -> wPlaceScale) = 0.3295 m
//
// WHAT WAS HERE BEFORE, AND WHY IT DREW NOTHING AT ALL. This line read 0.196035,
// derived as "a 0.19 m hand must occupy 6.46 map units: 6.46 / 32.9533". That is
// two compounding mistakes, and they multiply out exactly:
//
//   x34.0000  spends vr_vunits_per_meter a second time, when GetWeaponTransform
//             has already spent it -- the file header's own "100x class of bug".
//   x0.5766   targets 0.19 m, but the psprite hand these are meant to match is
//             0.3295 m: this mesh is a hand AND forearm (bbox runs -22.35 to
//             +10.60 on its long axis), not a bare palm.
//   ------
//   x19.6035  net. And 0.196035 / 0.01 = 19.6035. The two readings agree to six
//             figures, so nothing else is going on here.
//
// The symptom was NOT a big hand. 32.9533 * 0.196035 = 6.46 m, centred on a
// controller roughly half a metre from the face: the camera sits inside the mesh,
// and a surface seen from inside is backface-culled away. World hands read as
// "never drawn anywhere", indistinguishable from the feature being unwired. That
// is why it survived -- and why the tuning sliders could not find it either, since
// rs_hw_main_scale / rs_hw_off_scale bottom out at 0.05 (MENUDEF.txt:69, :78)
// while the correction needed was 1/19.6 = 0.051. Just out of reach.
//
// The main hand keeps the negative X. These are ONE mesh mirrored, and this path
// watches for a negative scale product to flip face culling to match; dropping
// the sign renders the main hand inside out.
// ===========================================================================

Model RS_HandWorldMain
{
	Path "models/hands"
	Model 0 "hand_left.iqm"
	PlacementCVars rs_hw_main
	// SKIN IS LOAD-BEARING, AND ITS ABSENCE IS SILENT.
	// IQMModel::RenderFrame (models_iqm.cpp:521-537) resolves a mesh skin in three
	// steps -- MODELDEF Skin, then SurfaceSkin, then the material name baked into
	// the IQM -- and if all three miss it hits "continue" and SKIPS THE MESH. Not a
	// white model, not a warning: no draw at all, nothing in the log.
	// hand_left.iqm names its material "aiStandardSurface1", a Maya default that
	// exists nowhere in this pk3, so these two blocks drew nothing from the day they
	// were written. Both psprite blocks above always carried a Skin, which is why
	// only the world hands were affected.
	Skin 0 "hand_basecolor.png"

	// PLACEMENT, FOUND IN A HEADSET 2026-08-30 AND FOLDED IN HERE.
	//
	// These are not guesses: they are the rs_hw_*_ofs_* / _pitch / _roll values
	// that were dialled in on the sliders until the mesh sat correctly on the
	// controller, transferred verbatim. That transfer is exact rather than
	// approximate -- ObjectToWorldMatrix sums the slider into the SAME translate
	// and rotate calls as these fields (models.cpp:855-861), which is the whole
	// reason the placement cvars were built to fold in rather than apply after.
	//
	// So the sliders go back to zero and nothing moves. Anyone tuning from here
	// starts from neutral instead of from someone else's correction.
	//
	// Both hands take the SAME numbers. They are one mesh mirrored on X, and a
	// mirror does not change which way is forward or up -- only the sign of the
	// scale, which stays on the Scale line where it belongs.
	//
	// Scale stayed at slider 1.0, so the derived 0.01 was right and is unchanged.
	Scale -0.01 0.01 0.01
	Offset 0.0 0.1 -0.05
	AngleOffset 0.0
	PitchOffset 90.0
	RollOffset -90.0
	NOAUTOREVERSE
	FollowMainHand
	BaseFrame
}

Model RS_GrabOvalMain
{
	Path "models"
	Model 0 "rs_wiresphere.obj"
	Skin  0 "rs_grab_idle.png"
	// Unit radius mesh, so the per-axis placement scale IS the semi-axis --
	// rs_grab_m_scale_x/_y/_z, the same three RS_Reach tests against.
	//
	// IN WHAT UNITS is the open question, and this comment used to answer it
	// flatly with "map units" while CVARINFO.txt answered it just as flatly with
	// "a unit-radius mesh arrives already metres across". Both cannot be right
	// and neither was ever checked against the headset. FollowMainHand means the
	// base transform is GetWeaponTransform, which carries a scale of
	// vr_vunits_per_meter -- so mesh space is metres and the CVARINFO reading is
	// the better-evidenced one. It is still a headset question, and the answer
	// now lives on rs_grab_scale_space rather than in an assertion in a comment.
	//
	// AND THE LETTERS ARE THE ENGINE'S, not ours: _scale_x scales the SIDEWAYS
	// axis here, because it multiplies the same matrix component _ofs_x
	// translates and vr_hand_ofs_x is captioned "Left / Right". The script test
	// had x on forward until 2026-08-26.
	Scale 1.0 1.0 1.0
	PlacementCVars rs_grab_m
	FollowMainHand
	FrameIndex RSGB A 0 0
	FrameIndex RSGB B 0 0
}

// ============================================================
// from RS_Holsters  (MODELDEF)
// ============================================================
Model RS_HolsterMarker
{
	Path "models"
	// ONE slot. Every slot with a model is drawn every frame -- a FrameIndex
	// selects a frame per slot, never which slot shows -- so a second
	// hot-skinned slot rendered on top of the idle one permanently, and on
	// top of the sphere when rs_holster_marker_shape asked for it. Hot and
	// idle are a skin swap on slot 0 (RS_HolsterMarker.SetHot).
	Model 0 "rs_holster_bracket.obj"
	Skin 0 "rs_wire_idle.png"
	// Unit-radius mesh, same convention rs_wiresphere.obj used -- this Scale
	// is what makes the reticle's true size match hsRadius exactly, the same
	// "what you see is the actual volume" property the sphere had.
	Scale 3.0 3.0 3.0

	// FLAGS MUST PRECEDE THE FrameIndex LINES. Each FrameIndex immediately
	// pushes a COPY of the in-progress model frame into the render table
	// (r_data/models.cpp: SpriteModelFrames.Push(smf), inside the FrameIndex
	// branch), capturing whatever flags are set AT THAT MOMENT. Flags written
	// after the last FrameIndex mutate a template nothing reads again -- they
	// parse without error and apply to nothing.
	USEACTORPITCH
	USEACTORROLL

	// Idle spin, entirely engine-driven -- no per-tic script needed. The
	// mesh is flat in its own local XY plane (every vertex z=0), so spinning
	// around local Z keeps the whole reticle in-plane, like a rotating dial,
	// rather than tumbling it out of the plane the player is meant to read it
	// in. Speed 0.2 is a full turn roughly every 1000 tics (~28s at 35
	// tics/sec) -- present, but not distracting.
	Rotating
	Rotation-Speed 0.2
	Rotation-Vector 0 0 1

	FrameIndex RSHM A 0 0
	FrameIndex RSHM B 0 0
}





// ============================================================
// from RS_Reload  (MODELDEF.txt)
// ============================================================
// THE AMMO IN YOUR OFF HAND. Five meshes, one per feed family, for every gun in
// every mod that will ever load.
//
// These blocks are copied in shape from RS_Hands' own hand blocks, because that
// is the path that is known to work on this renderer:
//
//   UseHandOffsets   take the controller transform, same as the hands do
//   NOAUTOREVERSE    no automatic frame reversal on a model with one frame
//   PitchOffset 90   the hands' correction; a mesh in the same hand needs it too
//   BaseFrame        LAST, and load-bearing. BaseFrame is the only thing that
//                    registers into BaseSpriteModelFrames and sets hasmodel,
//                    which is what a +DECOUPLEDANIMATIONS actor's model lookup
//                    reads. Without it the psprite falls through to the sprite
//                    path and draws nothing at all -- silently.
//
// NO FrameIndex ANYWHERE BELOW. FrameIndex and +DECOUPLEDANIMATIONS are mutually
// exclusive: a decoupled actor resolves through BaseSpriteModelFrames, which only
// BaseFrame populates. Set FrameIndex on one of these and it draws nothing, with
// nothing in the log.
//
// THE SCALES BELOW ARE MEASURED, NOT GUESSED. tools/md3_measure.py reads each
// mesh's own bounding box out of its MD3 frame record and divides a real-world
// target size by it:
//
//     scale = (target_metres * 173.44) / longest_axis_in_model_units
//
// 173.44 is what this HUD/psprite path works out to per metre. It is NOT 34 --
// that is the world path, and mixing the two is the whole "100x" class of bug.
//
//     mag_pistol    12.94 units  -> 0.12 m  -> 1.6089
//     shot_shell    10.56 units  -> 0.07 m  -> 1.1502
//     mag_plasma    17.18 units  -> 0.20 m  -> 2.0196
//     pod_rocket    21.85 units  -> 0.30 m  -> 2.3813
//     mag_chaingun  14.15 units  -> 0.22 m  -> 2.6972
//
// They are all different, which is why a single global slider was the wrong
// shape for this: these meshes were modelled at whatever scale each happened to
// be drawn at in Rusted Legacy, not to a common one. rr_ammo_scale stays as a
// multiplier ON TOP for taste, defaulting to 1.0 so it changes nothing.
//
// Off hand only, so no negative X mirror. RS_Hands mirrors its MAIN hand with a
// negative X scale because both hands share one mesh; nothing here is shared and
// nothing here is ever drawn on the main hand.

Model RR_AmmoMag
{
	Path "models/ammo"
	Model 0 "mag_pistol.md3"
	Skin 0 "berettam9.jpg"
	Scale 1.6089 1.6089 1.6089
	Offset 0.0 -18.0 -12.0
	UseHandOffsets
	NOAUTOREVERSE
	PitchOffset 90.0
	RollOffset 0.0
	AngleOffset 0.0
	BaseFrame
}
